Costruito con React

Tutto il sito gira su questo stack — niente CMS, niente page builder. Snippet veri dal repository, uno per tecnologia.

Leggi la documentazione— React
src/app/App.tsx
import { useState, useEffect } from "react";

export default function App() {
  const [scrolled, setScrolled] = useState(false);

  // La nav diventa "solida" dopo 60px di scroll
  useEffect(() => {
    const onScroll = () => setScrolled(window.scrollY > 60);
    window.addEventListener("scroll", onScroll, { passive: true });
    return () => window.removeEventListener("scroll", onScroll);
  }, []);

  return <Home nav={scrolled ? "solid" : "glass"} />;
}
